from pathlib import Path import re src = Path("/mnt/data/index(3).html").read_text(encoding="utf-8") # Extract the generated index.html and app.js from the uploaded source file. m_index = re.search(r"index = r'''(.*?)'''\n\n[L]?app = r'''", src, re.S) if not m_index: # More robust split around the known markers. start = src.index("index = r'''") + len("index = r'''") end = src.index("'''\n\napp = r'''", start) index_html = src[start:end] else: index_html = m_index.group(1) start = src.index("app = r'''") + len("app = r'''") end = src.index("'''\n\n[L]?out_dir", start) if "'''\n\n[L]?out_dir" in src else src.index("'''\n\nout_dir", start) app_js = src[start:end] # Fix the Rebirth response-field mismatch: # server.js returns { next: ... }, while app.js was reading data.next_rebirth. app_js = app_js.replace( "nextRebirth =\n data.next_rebirth || null;", "nextRebirth =\n data.next_rebirth || data.next || null;" ) # Also make the rebirth loader refresh the button state immediately after # a successful click and keep the current user data synchronized. app_js = app_js.replace( ' nextRebirth =\n data.next_rebirth || data.next || null;', ' nextRebirth =\n data.next_rebirth || data.next || null;' ) out_dir = Path("/mnt/data") (out_dir / "index.html").write_text(index_html, encoding="utf-8") (out_dir / "app.js").write_text(app_js, encoding="utf-8") print("Готово.") print("Созданы исправленные файлы:") print("/mnt/data/index.html") print("/mnt/data/app.js") print() print("Главное исправление: app.js теперь понимает ответ сервера `next`,") print("поэтому цена и кнопка Rebirth должны отображаться правильно.")
